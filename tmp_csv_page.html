    async function deepAnalyzeWithPipeline(rowObjects){
      try{
        const items = [];
        const take = Math.min(rowObjects.length, 5000);
        for(let i=0;i raw[k] ?? raw[k?.toLowerCase?.()] ?? raw[k?.toUpperCase?.()];
          const process_name = pick('process_name') || pick('Image') || pick('NewProcessName') || '';
          const file_path = pick('file_path') || pick('Path') || pick('Image') || '';
          const hash = pick('sha256') || pick('hash') || pick('md5') || '';
          const host = pick('host') || pick('Computer') || pick('Hostname') || '';
          const cmd = pick('CommandLine') || pick('command_line') || '';
          const item = { name: process_name||file_path||hash||('row_'+i), path: file_path, sha256: hash, host, command_line: cmd };
          // carry raw for potential context
          item.__raw = raw;
          items.push(item);
        }
        const body = { items };
        const resp = await fetch('/api/v1/artifacts/analyze_batch', { method:'POST', headers:{ 'Content-Type':'application/json', ...authHeaders() }, body: JSON.stringify(body) });
        const js = await resp.json();
        if(!resp.ok){ const msg = js && (js.detail||js.error) ? (js.detail||js.error) : ('pipeline_error:'+resp.status); _toast('Pipeline analyze failed: '+msg, 5000); throw new Error(msg); }
        const arts = (js && js.report && Array.isArray(js.report.all_artifacts)) ? js.report.all_artifacts : [];
        if(!arts.length){
          // Surface a clear banner when pipeline is unavailable in this environment
          try{ const b = document.getElementById('pipelineUnavailable'); if(b){ b.style.display='block'; } }catch(_){ }
          const meta = js && js.meta ? js.meta : (js && js.report && js.report.meta ? js.report.meta : null);
          const metaMsg = meta ? (typeof meta === 'string' ? meta : JSON.stringify(meta)) : '';
          throw new Error('No artifacts returned by pipeline' + (metaMsg ? (': '+metaMsg) : ''));
        }
        const list = arts.map(a=>{
          const rec = {
            process_name: a.name || '',
            file_path: a.path || '',
            hash: a.sha256 || '',
            host: a.host || '',
            verdict: (a.verdict||'').toUpperCase(),
            risk: (typeof a.risk==='number') ? a.risk : null,
            factors: Array.isArray(a.factors)? a.factors: [],
            raw: { analysis: { mitre: Array.isArray(a.mitre)? a.mitre: [] }, pipeline: a }
          };
          return rec;
        });
        // Instead of overwriting, merge pipeline results into existing LAST_RESULTS
        try{
          const existing = window.LAST_RESULTS || [];
          // Map by hash+host to replace or append
          const keyOf = (r)=> ((r.hash||'') + '|' + (r.host||'')).toLowerCase();
          const existingMap = {};
          existing.forEach((r,i)=> existingMap[keyOf(r)] = { r, i });
          for(const newRec of list){ const k = keyOf(newRec); if(existingMap[k]){ // merge fields
              const idx = existingMap[k].i; const old = existing[idx]; // merge arrays & fields
              old.factors = Array.from(new Set([...(old.factors||[]), ...(newRec.factors||[]), 'deep_enriched']));
              old.verdict = newRec.verdict || old.verdict;
              old.risk = (newRec.risk!=null) ? newRec.risk : old.risk;
              old.raw = Object.assign({}, old.raw||{}, newRec.raw||{});
              old._dread = computeDreadBreakdown(old.factors);
            } else {
              newRec.factors = Array.isArray(newRec.factors)? [...newRec.factors, 'deep_enriched'] : ['deep_enriched'];
              newRec._dread = computeDreadBreakdown(newRec.factors);
              existing.push(newRec);

    async function fetchRemainingPages(){
      const statusEl=document.getElementById('paginationStatus');
      const btnMore=document.getElementById('btnFetchMore');
      const btnAbort=document.getElementById('btnAbortFetch');
      PAGINATION_ABORT=false;
      try{
        const sessionsJson = localStorage.getItem('lastTabularSessions');
        if(!sessionsJson){ statusEl.textContent='No sessions stored'; return; }
        const sessions = JSON.parse(sessionsJson);
        if(!sessions || !sessions.length){ statusEl.textContent='No sessions available'; return; }
        const session = sessions[0];
        statusEl.textContent='Enumerating pages...';
        btnMore.disabled=true; btnAbort.style.display='inline-block';
        // Determine total rows via summary endpoint
        let totalRows=null; let headers=[];
        try{
          const sumR = await fetch(`/api/v1/upload/tabular/summary?session=${encodeURIComponent(session)}`, { headers:{ ...authHeaders() } });
          if(sumR.ok){ const sumJ=await sumR.json(); totalRows=sumJ.total_rows; headers=sumJ.headers||[]; }
          else { try{ const sd = await sumR.json(); _toast('Summary fetch failed: '+(sd.detail||sd.error||sumR.status),4000); }catch(_){ _toast('Summary fetch failed: '+sumR.status,4000); } }
        }catch(e){ _toast('Summary fetch failed: '+(e&&e.message||e),4000); }
        if(totalRows==null){ statusEl.textContent='Unable to fetch summary'; btnMore.disabled=false; btnAbort.style.display='none'; return; }
        const existing = window.LAST_RESULTS||[]; const existingCount=existing.length;
        if(existingCount >= totalRows){ statusEl.textContent=`All ${totalRows} rows already loaded.`; btnMore.disabled=false; btnAbort.style.display='none'; return; }
        if(totalRows > 75000){
          // Non-blocking confirm: show a toast and require explicit click on Fetch to continue in UI.
          try{
            const msg = `This file has ${totalRows} rows; loading all may impact memory. Click 'Fetch Remaining Pages' to continue.`;
            const s = statusEl; if(s) s.textContent = msg;
            // show a visual toast via shared notifications if present
            try{ if(window.notifications && window.notifications.showToast) window.notifications.showToast(msg, 8000); }catch(_){}
            btnMore.disabled = false; btnAbort.style.display='none';
            return;
          }catch(e){ /* fallback to abort */ statusEl.textContent='Aborted by user'; btnMore.disabled=false; btnAbort.style.display='none'; return; }
        }
        let offset = existingCount; const batch = 1000; let appended=0; const startTs=Date.now();
        while(offset < totalRows){
          if(PAGINATION_ABORT){ statusEl.textContent = `Aborted after ${appended} rows appended.`; break; }
            statusEl.textContent = `Fetching rows ${offset} - ${Math.min(offset+batch,totalRows)} / ${totalRows} ...`;
          try{
            const r = await fetch(`/api/v1/upload/tabular/page?session=${encodeURIComponent(session)}&offset=${offset}&limit=${batch}`, { headers:{ ...authHeaders() } });
            if(!r.ok){ statusEl.textContent=`Page fetch failed (${r.status})`; break; }
            const j = await r.json();
            const rows = (j.rows||[]).map(arr=>{ const obj={}; (j.headers||headers||[]).forEach((h,i)=> obj[h]=arr[i]); return normalizeRow(obj); });
            if(!rows.length){ break; }
            window.LAST_RESULTS = window.LAST_RESULTS.concat(rows);
            appended += rows.length; offset += rows.length;
            if(offset >= totalRows || j.complete){ break; }
          }catch(e){ statusEl.textContent='Error: '+(e.message||e); break; }
          // Yield to UI
          await new Promise(r=> setTimeout(r, 30));
        }
        recalcIndex(); renderResults({results:window.LAST_RESULTS, total_rows:window.LAST_RESULTS.length});
        renderQuickHunts(window.LAST_RESULTS); renderHostInsights(window.LAST_RESULTS);
        const dur=((Date.now()-startTs)/1000).toFixed(1);
        if(!PAGINATION_ABORT) statusEl.textContent = `Loaded ${window.LAST_RESULTS.length}/${totalRows} rows in ${dur}s (added ${appended}).`;
      }finally{ const btnMore=document.getElementById('btnFetchMore'); const btnAbort=document.getElementById('btnAbortFetch'); if(btnMore) btnMore.disabled=false; if(btnAbort) btnAbort.style.display='none'; }
    }
